Replace the hard gate on line 243 (`if cycle["state"] != WheelState.SHORT_PUT.value: return None`) with branching:

```python
state = cycle["state"]

if state == WheelState.SHORT_PUT.value:
    # ... existing put logic (lines 250-364), unchanged ...

elif state == WheelState.COVERED_CALL.value:
    # Get open CALL position
    positions = self.db.get_cycle_positions(cycle["id"])
    open_positions = [p for p in positions if p["status"] == "OPEN"]
    if not open_positions:
        logger.warning("COVERED_CALL cycle %d has no open positions", cycle["id"])
        return None
    call_pos = open_positions[-1]  # most recent if multiple (shouldn't happen)

    # Check expiry
    expiry = date.fromisoformat(call_pos["expiry_date"])
    today = get_et_now().date()
    if today <= expiry:
        return None  # not expired yet

    # Check if call still live in E*TRADE (T-03-13: wrap API)
    try:
        live_options = self.client.get_options_positions(self.account_id_key)
    except Exception as exc:
        logger.error("detect_and_process_expiry: get_options_positions failed -- %s", exc)
        return None
    live_symbols = {p["symbol"] for p in live_options}
    if call_pos["symbol"] in live_symbols:
        return None  # settlement not complete

    # Check if IBIT shares still present
    try:
        equity_positions = self.client.get_account_positions(self.account_id_key)
    except Exception as exc:
        logger.error("detect_and_process_expiry: get_account_positions failed -- %s", exc)
        return None

    ibit_shares = [
        p for p in equity_positions
        if p.get("Product", {}).get("symbol") == "IBIT"
        and p.get("Product", {}).get("securityType") == "EQ"
    ]

    if not ibit_shares:
        # CALLED AWAY: call gone AND shares gone
        # Full-cycle P&L
        put_premium = cycle["put_premium_received"] * 100
        call_premiums = (cycle.get("covered_call_premiums_collected") or 0.0) * 100
        call_strike = float(call_pos["strike"])
        original_cost_basis_per_share = cycle["put_strike"] - cycle["put_premium_received"]
        shares_pnl = (call_strike - original_cost_basis_per_share) * 100
        total_pnl = put_premium + call_premiums + shares_pnl
        # Simpler: total_pnl = (call_strike - cycle["put_strike"]) * 100 + put_premium + call_premiums
        # But be explicit.

        self.db.transition_wheel_state(
            cycle["id"], WheelState.CASH, "called_away",
            realized_pnl=total_pnl,
        )
        self.db.close_wheel_position(call_pos["id"], close_premium=0.0)
        self.db.log_event("INFO", "called_away", {
            "cycle_id": cycle["id"],
            "call_strike": call_strike,
            "total_pnl": total_pnl,
        })
        return "called_away"
    else:
        # OTM CALL EXPIRY: call gone, shares remain
        self.db.transition_wheel_state(
            cycle["id"], WheelState.HOLDING_SHARES, "call_expired_otm",
        )
        self.db.close_wheel_position(call_pos["id"], close_premium=0.0)
        self.db.log_event("INFO", "call_expired_otm", {
            "cycle_id": cycle["id"],
        })
        return "call_expired_otm"
else:
    # Not a state we handle (e.g., CASH) -- skip
    return None
```
