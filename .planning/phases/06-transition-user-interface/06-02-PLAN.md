       class WheelCommandsMixin:
           """Mixin for /wheel and /wheelmode Telegram commands (TR-02, TR-03)."""

           async def _cmd_wheel(self: "TelegramBot", update: Update, context):
               """Show current wheel cycle status, positions, and P&L."""
               if not self._is_authorized(update):
                   await self._send_unauthorized_response(update)
                   return

               from ..database import get_database
               db = get_database()

               cycle = db.get_active_cycle()
               if cycle is None:
                   await update.message.reply_text("No active wheel cycle.")
                   return

               positions = db.get_cycle_positions(cycle["id"])
               open_positions = [p for p in positions if p["status"] == "OPEN"]

               state = cycle["state"]
               cost_basis = cycle.get("cost_basis", 0.0) or 0.0
               total_premium = (
                   (cycle.get("put_premium_received") or 0.0)
                   + (cycle.get("covered_call_premiums_collected") or 0.0)
               ) * 100  # per-share to total (100 shares)

               lines = [f"*WHEEL CYCLE -- {state}*\n"]
               lines.append(f"Cost Basis: ${cost_basis:.2f}/share")
               lines.append(f"Total Premium: ${total_premium:.2f}")

               if open_positions:
                   lines.append("\n*Open Positions:*")
                   for p in open_positions:
                       from datetime import date
                       from ..utils import get_et_now
                       expiry = date.fromisoformat(p["expiry_date"])
                       dte = max(0, (expiry - get_et_now().date()).days)
                       pnl_str = ""
                       if p.get("premium_received") and p.get("current_value") is not None:
                           pnl = (p["premium_received"] - (p.get("current_value") or 0)) * 100
                           pnl_str = f" | P&L: ${pnl:.2f}"
                       lines.append(
                           f"  {p['option_type']} ${p['strike']:.0f} exp {p['expiry_date']} "
                           f"({dte} DTE) | delta={p.get('delta', 'N/A')}{pnl_str} | "
                           f"premium=${p['premium_received']:.2f}"
                       )
                   lines.append("\n_Greeks shown are entry values._")

               await update.message.reply_text("\n".join(lines), parse_mode="Markdown")

       def _job_wheel_daily_summary(self) -> None:
           """Send wheel position summary at 4:30 PM ET (TR-05)."""
           now = get_et_now()
           if not is_trading_day(now.date()):
               return

           try:
               cycle = self.db.get_active_cycle()
               if not cycle:
                   self._send_notification("No active wheel positions today.")
                   return

               positions = self.db.get_cycle_positions(cycle["id"])
               open_positions = [p for p in positions if p["status"] == "OPEN"]

               state = cycle["state"]
               cost_basis = cycle.get("cost_basis", 0.0) or 0.0
               total_premium = (
                   (cycle.get("put_premium_received") or 0.0)
                   + (cycle.get("covered_call_premiums_collected") or 0.0)
               ) * 100

               lines = ["*WHEEL DAILY SUMMARY*\n"]
               lines.append(f"Cycle State: {state}")
               lines.append(f"Cost Basis: ${cost_basis:.2f}/share")
               lines.append(f"Total Premium: ${total_premium:.2f}")

               if open_positions:
                   lines.append("\n*Positions:*")
                   for p in open_positions:
                       from datetime import date
                       expiry = date.fromisoformat(p["expiry_date"])
                       dte = max(0, (expiry - now.date()).days)
                       max_risk = p["strike"] * 100

                           f"({dte} DTE) | max risk: ${max_risk:,.0f} | "
                           f"premium: ${p['premium_received']:.2f}"
                       )
               else:
                   lines.append("\nNo open positions.")

               self._send_notification("\n".join(lines))

           except Exception as e:
               logger.error(f"Wheel daily summary failed: {e}")
               self._error_count += 1
